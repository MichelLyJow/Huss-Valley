local function flat(v) return Vector3.new(v.X, 0, v.Z) end
local function getCharacter(player) return (player or LocalPlayer).Character end
local function getHumanoid(player) local c = getCharacter(player) return c and c:FindFirstChildOfClass("Humanoid") end
local function getRoot(player) local c = getCharacter(player) return c and c:FindFirstChild("HumanoidRootPart") end
local function getRole() return LocalPlayer:GetAttribute("GameRole") or "Lobby" end
local function getRunState() return LocalPlayer:GetAttribute("RunState") or "Idle" end

local function isRunnerActive()
    return LocalPlayer:GetAttribute("InMatch") == true and getRole() == "Runner" and getRunState() == "Active"
end

local function isCatcherActive()
    return LocalPlayer:GetAttribute("InMatch") == true and getRole() == "Catcher" and getRunState() == "Active"
end

local function canMove()
    if LocalPlayer:GetAttribute("ClientReady") ~= true or Session:GetAttribute("GlobalPaused") == true or Session:GetAttribute("MapChanging") == true then
        return false
    end
    local c = getCharacter()
    if not c or c:GetAttribute("Ragdolled") == true or c:GetAttribute("GearMotion") == true or c:GetAttribute("MovementLocked") == true then
        return false
    end
    local h = getHumanoid()
    return h and h.Health > 0 and not h.PlatformStand and not h.Sit
end

-- Game Event Listeners
SpinWheelEvent.OnClientEvent:Connect(function(kind, payload)
    if kind == "State" and type(payload) == "table" then
        runtime.wheelCredits = tonumber(payload.credits) or 0
        if SETTINGS.autoSpinWheel and LocalPlayer:GetAttribute("InMatch") ~= true then
            if runtime.wheelCredits >= SETTINGS.wheelMinimumCredits and not runtime.wheelSpinning then
                runtime.wheelSpinning = true
                pcall(function() SpinWheelEvent:FireServer("Spin") end)
                task.delay(1, function() runtime.wheelSpinning = false end)
            end
        end
    end
end)

-- Main Execution Engine
RunService.RenderStepped:Connect(function()
    local character = getCharacter()
    local hum = getHumanoid()
    local root = getRoot()

    if not character or not hum or hum.Health <= 0 or not root then return end

    local role = getRole()
    local nativeOutput = tonumber(character:GetAttribute("MovementSpeed")) or 16
    if role == "Runner" and SETTINGS.customSpeed then
        hum.WalkSpeed = nativeOutput * math.clamp(SETTINGS.speedMultiplier, 1, 2)
    elseif role == "Catcher" and SETTINGS.catcherCustomSpeed then
        hum.WalkSpeed = nativeOutput * math.clamp(SETTINGS.catcherSpeedMultiplier, 1, 2)
    end

    -- Runner Auto Target Logic
    if isRunnerActive() and SETTINGS.runnerEnabled and canMove() then
        if SETTINGS.autoRevive then
            for _, player in ipairs(Players:GetPlayers()) do
                if player ~= LocalPlayer and player.Character then
                    local pRoot = getRoot(player)
                    if pRoot and player.Character:GetAttribute("RescueAvailable") == true then
                        if (pRoot.Position - root.Position).Magnitude <= RESCUE_RANGE then
                            pcall(function() RescueEvent:FireServer("Begin", player.UserId) end)
                            pcall(function() RescueEvent:FireServer("Finish", player.UserId) end)
                        end
                    end
                end
            end
        end
    end

    -- Catcher Attack Logic
    if isCatcherActive() and canMove() then
        for _, player in ipairs(Players:GetPlayers()) do
            if player ~= LocalPlayer and player:GetAttribute("GameRole") == "Runner" and player.Character then
                local tRoot = getRoot(player)
                if tRoot then
                    local dist = (tRoot.Position - root.Position).Magnitude
                    if SETTINGS.autoChase then
                        hum:MoveTo(tRoot.Position)
                    end
                    if SETTINGS.autoTackle and dist <= SETTINGS.tackleRange then
                        pcall(function() TacklePrediction.request() end)
                    end
                    if SETTINGS.autoMelee and dist <= SETTINGS.meleeRange then
                        runtime.meleeId += 1
                        pcall(function() MeleeEvent:FireServer(runtime.meleeId) end)
                    end
                end
            end
        end
    end

    RunnerStatusParagraph:SetTitle("Runner: " .. tostring(role))
    RunnerStatusParagraph:SetDesc(string.format("State: %s\nGoal: %s", tostring(getRunState()), tostring(runtime.goalMode)))
    
    CatcherStatusParagraph:SetTitle("Catcher: " .. tostring(role))
    CatcherStatusParagraph:SetDesc(string.format("State: %s\nTarget: %s", tostring(getRunState()), runtime.catcherTarget and runtime.catcherTarget.Name or "None"))
end)
